CREATE DATABASE IF NOT EXISTS mercaDaw;
USE mercaDaw;

-- Tabla de tipos de producto
CREATE TABLE tipos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL UNIQUE
);

-- Tabla de productos
CREATE TABLE products (
    id INT AUTO_INCREMENT PRIMARY KEY,
    referencia VARCHAR(50) NOT NULL UNIQUE,
    tipo INT,
    nombre VARCHAR(100),
    descripcion VARCHAR(300),
    cantidad INT,
    precio DOUBLE,
    descuento INT,
    iva INT,
    aplicarDto BOOLEAN,

    FOREIGN KEY (tipo) REFERENCES tipos(id)
);
